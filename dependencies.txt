# System packages a development machine needs, one per line. Blank lines and
# anything after a # are ignored. The devcontainer installs these at image
# build time; a bare host developing on this project needs the same set.
#
# The server image needs a shorter list and reads docker/dependencies.txt. The
# two overlap in the transports and diverge everywhere else: nothing here that
# builds, runs or displays the desktop app belongs in a container that serves
# an API.

# `make stop` finds the server by the port it listens on.
lsof

# ssd, ethernet, hardware_trigger reach a unit over SSH.
openssh-client

# can_bus. python-can drives SocketCAN through the kernel directly; these
# bring an interface up and watch frames by hand.
can-utils
iproute2

# piezo. paho-mqtt speaks MQTT directly; these probe a broker by hand.
mosquitto-clients

# The DAQ is claimed over raw USB, which pyusb reaches by dlopening libusb at
# import time rather than linking it. lsusb is how an instrument missing from
# the bus is told apart from one present and unopenable.
libusb-1.0-0
usbutils

# `make app-build` writes an AppImage, which mounts itself through FUSE to run.
# Without this it exits with "dlopen(): error loading libfuse.so.2", and
# --appimage-extract-and-run is the way around it.
libfuse2

# Electron's own runtime. `make app-dev` and the built app both fail to start
# without these. The list matches deb.depends in app/electron-builder.json, so
# running from the checkout needs no more than installing the package would.
libasound2
libatspi2.0-0
libgbm1
libgtk-3-0
libnotify4
libnss3
libsecret-1-0
libuuid1
libxss1
libxtst6
xdg-utils

# Showing that window. The devcontainer forwards the host's display through the
# X11 socket it mounts, and xauth is what makes the cookie on it usable. Xvfb
# stands in where there is no display at all, which is how the app is exercised
# in a headless container.
xauth
xvfb

# Rasterises frontend/public/favicon.svg into app/icons/icon.png, the icon
# electron-builder packages. The PNG is checked in; this regenerates it.
librsvg2-bin
